-- ============================================
-- MIGRATION: Schema RSVP App (v2 — izin diperluas)
-- ============================================

drop table if exists registrations cascade;
drop table if exists events cascade;
drop table if exists admins cascade;
drop function if exists validate_event_code;

create extension if not exists "pgcrypto";

-- ============================================
-- 1. Tabel admin
-- ============================================
create table admins (
  id uuid primary key references auth.users(id) on delete cascade,
  nama text,
  created_at timestamptz default now()
);

-- ============================================
-- 2. Tabel event
-- Tambahan: jam_acara dan deskripsi
-- catering_options: array JSON berisi {id, name, image_url, price, category}
-- category: 'food' atau 'drink' (opsional, default dianggap 'food' di frontend
-- kalau tidak diisi — dipakai untuk mengelompokkan tampilan menu saat RSVP)
-- ============================================
create table events (
  id uuid primary key default gen_random_uuid(),
  nama_acara text not null,
  deskripsi text,
  kode_unik text not null unique,
  tanggal_acara date,
  jam_acara time,
  tempat text,
  catering_options jsonb not null default '[]'::jsonb,
  qris_url text,   -- URL QRIS pembayaran, diupload admin lewat halaman Data Registrasi
  created_by uuid references admins(id),
  created_at timestamptz default now()
);

-- ============================================
-- 3. Tabel registrasi
-- jenis: kategori utama (member / pengurus / izin)
-- Data diri (nama, nim, fakultas, jurusan, angkatan, jabatan) seragam dan
-- wajib diisi untuk semua jenis pendaftaran:
--  - angkatan: tahun angkatan, wajib untuk semua jenis
--  - jabatan: otomatis 'Member' kalau jenis=member, diisi manual untuk
--    pengurus/izin (izin tetap harus menyatakan jabatannya, atau 'Member'
--    kalau dia member biasa)
-- ============================================
create table registrations (
  id uuid primary key default gen_random_uuid(),
  event_id uuid not null references events(id) on delete cascade,

  jenis text not null check (jenis in ('member', 'pengurus', 'izin')),

  nama text not null,
  nim text not null,
  fakultas text not null,
  jurusan text not null,
  angkatan text not null,
  jabatan text not null,

  alasan_tidak_hadir text,     -- wajib kalau jenis = izin
  bukti_perizinan_url text,    -- wajib kalau jenis = izin (menggantikan bukti SG invitation)

  catering_choices jsonb,      -- wajib kalau jenis = member/pengurus (hadir)
  total_harga numeric(10,2),
  bukti_bayar_url text,        -- wajib kalau jenis = member/pengurus (hadir)

  created_at timestamptz default now(),

  -- Validasi jalur HADIR (member/pengurus): wajib pilih catering + bukti bayar
  constraint check_hadir_lengkap check (
    jenis = 'izin'
    or (
      catering_choices is not null
      and jsonb_array_length(catering_choices) > 0
      and bukti_bayar_url is not null
    )
  ),

  -- Validasi jalur IZIN: wajib isi alasan dan bukti perizinan
  constraint check_izin_lengkap check (
    jenis != 'izin'
    or (
      alasan_tidak_hadir is not null
      and bukti_perizinan_url is not null
    )
  )
);

create index idx_events_kode on events(kode_unik);
create index idx_registrations_event on registrations(event_id);

-- ============================================
-- ROW LEVEL SECURITY
-- ============================================

alter table admins enable row level security;
alter table events enable row level security;
alter table registrations enable row level security;

create policy "admin manage own profile" on admins
  for all using (auth.uid() = id);

create policy "admin manage own events" on events
  for all using (auth.uid() = created_by);

create policy "public insert registration" on registrations
  for insert with check (true);

create policy "admin read own registrations" on registrations
  for select using (
    exists (
      select 1 from events
      where events.id = registrations.event_id
      and events.created_by = auth.uid()
    )
  );

-- ============================================
-- RPC FUNCTION: validasi kode unik
-- ============================================

create or replace function validate_event_code(input_code text)
returns table(
  event_id uuid,
  nama_acara text,
  deskripsi text,
  tanggal_acara date,
  jam_acara time,
  tempat text,
  catering_options jsonb
)
language sql
security definer
as $$
  select id, nama_acara, deskripsi, tanggal_acara, jam_acara, tempat, catering_options
  from events
  where kode_unik = input_code;
$$;

-- ============================================
-- MIGRATION: Data diri seragam untuk semua jenis
-- (nama, nim, fakultas, jurusan, angkatan, jabatan sekarang wajib
-- diisi untuk member/pengurus/izin — identitas_izin tidak dipakai lagi)
-- ============================================

-- 1. Tambah kolom fakultas
alter table registrations add column if not exists fakultas text;
update registrations set fakultas = 'Belum diisi' where fakultas is null;
alter table registrations alter column fakultas set not null;

-- 2. Isi nilai default untuk baris lama yang belum punya angkatan/jabatan,
-- lalu wajibkan kedua kolom untuk semua baris baru
update registrations set angkatan = coalesce(angkatan, '-') where angkatan is null;
update registrations set jabatan = coalesce(jabatan, case when jenis = 'member' then 'Member' else '-' end) where jabatan is null;
alter table registrations alter column angkatan set not null;
alter table registrations alter column jabatan set not null;

-- 3. identitas_izin sudah tidak relevan (semua jenis kini mengisi field yang sama)
alter table registrations drop constraint if exists check_field_identitas;
alter table registrations drop constraint if exists check_izin_lengkap;
alter table registrations add constraint check_izin_lengkap check (
  jenis != 'izin' or (alasan_tidak_hadir is not null and bukti_perizinan_url is not null)
);
alter table registrations drop column if exists identitas_izin;

-- ============================================
-- MIGRATION: Tambah field tempat (venue) acara
-- ============================================

alter table events add column if not exists tempat text;

drop function if exists validate_event_code(text);
create or replace function validate_event_code(input_code text)
returns table(
  event_id uuid,
  nama_acara text,
  deskripsi text,
  tanggal_acara date,
  jam_acara time,
  tempat text,
  catering_options jsonb
)
language sql
security definer
as $$
  select id, nama_acara, deskripsi, tanggal_acara, jam_acara, tempat, catering_options
  from events
  where kode_unik = input_code;
$$;
